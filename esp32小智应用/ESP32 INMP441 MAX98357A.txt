/**
 * ESP32 + INMP441 (麦克风) + MAX98357A (功放) 综合测试
 * 引脚定义：
 *   麦克风: WS=25, SCK=26, SD=32, L/R=GND
 *   功放:   BCLK=14, LRC=27, DIN=33, SD(使能)=12(可选)
 * 
 * 测试模式: 通过 TEST_MODE 宏选择
 *   1 = 功放测试 (播放 1kHz 正弦波)
 *   2 = 麦克风测试 (串口打印音量条)
 *   3 = 实时回环 (麦克风 -> 功放)
 */

#include <Arduino.h>
#include <driver/i2s.h>

// ========== 引脚定义 ==========
// 麦克风 (I2S RX)
#define I2S_MIC_WS     25
#define I2S_MIC_SCK    26
#define I2S_MIC_SD     32

// 功放 (I2S TX)
#define I2S_SPK_BCLK   14
#define I2S_SPK_LRC    27
#define I2S_SPK_DIN    33

// 功放使能引脚 (可选，如果未连接则注释或保留但不用)
// #define PA_ENABLE      12

// ========== 音频参数 ==========
#define SAMPLE_RATE    16000
#define DMA_BUF_LEN    256

// ========== 测试模式选择 ==========
// 修改下面的数字: 1=功放测试, 2=麦克风测试, 3=回环测试
#define TEST_MODE      3

// ========== 功放测试 ==========
#if TEST_MODE == 1
void setup() {
    Serial.begin(115200);
    delay(1000);
    Serial.println("===== 功放测试 (1kHz 正弦波) =====");
    Serial.println("接线: 功放 BCLK=14, LRC=27, DIN=33, VIN=5V, GND");
    Serial.println("如果无声, 请检查 SD 引脚是否为高电平 (可接 3.3V 或 GPIO12 拉高)");
    
    #ifdef PA_ENABLE
    pinMode(PA_ENABLE, OUTPUT);
    digitalWrite(PA_ENABLE, HIGH);
    #endif

    i2s_config_t spkConfig = {
        .mode = (i2s_mode_t)(I2S_MODE_MASTER | I2S_MODE_TX),
        .sample_rate = SAMPLE_RATE,
        .bits_per_sample = I2S_BITS_PER_SAMPLE_16BIT,
        .channel_format = I2S_CHANNEL_FMT_RIGHT_LEFT,
        .communication_format = I2S_COMM_FORMAT_STAND_I2S,
        .intr_alloc_flags = ESP_INTR_FLAG_LEVEL1,
        .dma_buf_count = 8,
        .dma_buf_len = DMA_BUF_LEN,
        .use_apll = false,
        .tx_desc_auto_clear = true,
        .fixed_mclk = -1
    };
    i2s_pin_config_t spkPins;
    spkPins.bck_io_num = I2S_SPK_BCLK;
    spkPins.ws_io_num = I2S_SPK_LRC;
    spkPins.data_out_num = I2S_SPK_DIN;
    spkPins.data_in_num = I2S_PIN_NO_CHANGE;
    spkPins.mck_io_num = I2S_PIN_NO_CHANGE;

    if (i2s_driver_install(I2S_NUM_0, &spkConfig, 0, NULL) != ESP_OK) {
        Serial.println("I2S 安装失败");
        while(1);
    }
    if (i2s_set_pin(I2S_NUM_0, &spkPins) != ESP_OK) {
        Serial.println("引脚设置失败");
        while(1);
    }
    Serial.println("开始播放 1kHz 正弦波...");
}

void loop() {
    int16_t buffer[DMA_BUF_LEN];
    static float phase = 0;
    for (int i = 0; i < DMA_BUF_LEN; i++) {
        buffer[i] = (int16_t)(32767 * 0.5 * sin(phase));
        phase += 2 * PI * 1000 / SAMPLE_RATE;
        if (phase >= 2*PI) phase -= 2*PI;
    }
    size_t bytesWritten;
    i2s_write(I2S_NUM_0, (const char*)buffer, sizeof(buffer), &bytesWritten, portMAX_DELAY);
}
#endif

// ========== 麦克风测试 ==========
#if TEST_MODE == 2
void setup() {
    Serial.begin(115200);
    delay(1000);
    Serial.println("===== 麦克风测试 (音量条) =====");
    Serial.println("接线: 麦克风 WS=25, SCK=26, SD=32, L/R=GND, VDD=3.3V");
    Serial.println("对着麦克风说话, 串口将显示音量条");

    i2s_config_t micConfig = {
        .mode = (i2s_mode_t)(I2S_MODE_MASTER | I2S_MODE_RX),
        .sample_rate = SAMPLE_RATE,
        .bits_per_sample = I2S_BITS_PER_SAMPLE_16BIT,
        .channel_format = I2S_CHANNEL_FMT_RIGHT_LEFT,
        .communication_format = I2S_COMM_FORMAT_STAND_I2S,
        .intr_alloc_flags = ESP_INTR_FLAG_LEVEL1,
        .dma_buf_count = 8,
        .dma_buf_len = DMA_BUF_LEN,
        .use_apll = false,
        .fixed_mclk = -1
    };
    i2s_pin_config_t micPins;
    micPins.bck_io_num = I2S_MIC_SCK;
    micPins.ws_io_num = I2S_MIC_WS;
    micPins.data_out_num = I2S_PIN_NO_CHANGE;
    micPins.data_in_num = I2S_MIC_SD;
    micPins.mck_io_num = I2S_PIN_NO_CHANGE;

    if (i2s_driver_install(I2S_NUM_1, &micConfig, 0, NULL) != ESP_OK) {
        Serial.println("I2S 安装失败");
        while(1);
    }
    if (i2s_set_pin(I2S_NUM_1, &micPins) != ESP_OK) {
        Serial.println("引脚设置失败");
        while(1);
    }
    Serial.println("开始采集...");
}

void loop() {
    int16_t buffer[DMA_BUF_LEN * 2];
    size_t bytesRead;
    if (i2s_read(I2S_NUM_1, buffer, sizeof(buffer), &bytesRead, portMAX_DELAY) != ESP_OK) {
        return;
    }
    int samples = bytesRead / sizeof(int16_t);
    long sumAbs = 0;
    for (int i = 0; i < samples / 2; i++) {
        sumAbs += abs(buffer[i * 2]);   // 左声道 (L/R=GND)
    }
    int avg = sumAbs / (samples / 2);
    int bars = constrain(map(avg, 0, 20000, 0, 40), 0, 40);
    Serial.print("[");
    for (int i=0; i<bars; i++) Serial.print("#");
    for (int i=bars; i<40; i++) Serial.print(" ");
    Serial.println("]");
    delay(50);
}
#endif

// ========== 回环测试 ==========
#if TEST_MODE == 3
void setup() {
    Serial.begin(115200);
    delay(1000);
    Serial.println("===== 实时回环测试 (麦克风 -> 功放) =====");
    Serial.println("接线检查:");
    Serial.println("  麦克风: WS=25, SCK=26, SD=32, L/R=GND, VDD=3.3V");
    Serial.println("  功放:   BCLK=14, LRC=27, DIN=33, VIN=5V");
    Serial.println("如果功放无声, 请将 SD 引脚接 3.3V 或 GPIO12 拉高");
    
    #ifdef PA_ENABLE
    pinMode(PA_ENABLE, OUTPUT);
    digitalWrite(PA_ENABLE, HIGH);
    Serial.println("功放使能引脚已拉高");
    #endif

    // 卸载残留驱动
    i2s_driver_uninstall(I2S_NUM_0);
    i2s_driver_uninstall(I2S_NUM_1);
    delay(10);

    // ----- 功放 I2S (TX) -----
    i2s_config_t spkConfig = {
        .mode = (i2s_mode_t)(I2S_MODE_MASTER | I2S_MODE_TX),
        .sample_rate = SAMPLE_RATE,
        .bits_per_sample = I2S_BITS_PER_SAMPLE_16BIT,
        .channel_format = I2S_CHANNEL_FMT_RIGHT_LEFT,
        .communication_format = I2S_COMM_FORMAT_STAND_I2S,
        .intr_alloc_flags = ESP_INTR_FLAG_LEVEL1,
        .dma_buf_count = 8,
        .dma_buf_len = DMA_BUF_LEN,
        .use_apll = false,
        .tx_desc_auto_clear = true,
        .fixed_mclk = -1
    };
    i2s_pin_config_t spkPins;
    spkPins.bck_io_num = I2S_SPK_BCLK;
    spkPins.ws_io_num = I2S_SPK_LRC;
    spkPins.data_out_num = I2S_SPK_DIN;
    spkPins.data_in_num = I2S_PIN_NO_CHANGE;
    spkPins.mck_io_num = I2S_PIN_NO_CHANGE;

    // ----- 麦克风 I2S (RX) -----
    i2s_config_t micConfig = {
        .mode = (i2s_mode_t)(I2S_MODE_MASTER | I2S_MODE_RX),
        .sample_rate = SAMPLE_RATE,
        .bits_per_sample = I2S_BITS_PER_SAMPLE_16BIT,
        .channel_format = I2S_CHANNEL_FMT_RIGHT_LEFT,
        .communication_format = I2S_COMM_FORMAT_STAND_I2S,
        .intr_alloc_flags = ESP_INTR_FLAG_LEVEL1,
        .dma_buf_count = 8,
        .dma_buf_len = DMA_BUF_LEN,
        .use_apll = false,
        .fixed_mclk = -1
    };
    i2s_pin_config_t micPins;
    micPins.bck_io_num = I2S_MIC_SCK;
    micPins.ws_io_num = I2S_MIC_WS;
    micPins.data_out_num = I2S_PIN_NO_CHANGE;
    micPins.data_in_num = I2S_MIC_SD;
    micPins.mck_io_num = I2S_PIN_NO_CHANGE;

    // 安装功放
    if (i2s_driver_install(I2S_NUM_0, &spkConfig, 0, NULL) != ESP_OK) {
        Serial.println("功放 I2S 安装失败");
        while(1);
    }
    if (i2s_set_pin(I2S_NUM_0, &spkPins) != ESP_OK) {
        Serial.println("功放引脚设置失败");
        while(1);
    }
    // 安装麦克风
    if (i2s_driver_install(I2S_NUM_1, &micConfig, 0, NULL) != ESP_OK) {
        Serial.println("麦克风 I2S 安装失败");
        while(1);
    }
    if (i2s_set_pin(I2S_NUM_1, &micPins) != ESP_OK) {
        Serial.println("麦克风引脚设置失败");
        while(1);
    }
    Serial.println("初始化成功, 开始实时回环...");
    Serial.println("对着麦克风说话, 扬声器应听到自己的声音");
}

void loop() {
    int16_t micBuffer[DMA_BUF_LEN * 2];
    size_t bytesRead;
    if (i2s_read(I2S_NUM_1, micBuffer, sizeof(micBuffer), &bytesRead, portMAX_DELAY) != ESP_OK) {
        return;
    }
    int16_t spkBuffer[DMA_BUF_LEN * 2];
    int numSamples = bytesRead / sizeof(int16_t);
    for (int i = 0; i < numSamples / 2; i++) {
        int16_t left = micBuffer[i * 2];   // 左声道
        // 可选增益 (如果音量太小, 取消注释并调整倍数)
        // left = constrain(left * 2, -32768, 32767);
        spkBuffer[i * 2] = left;
        spkBuffer[i * 2 + 1] = left;
    }
    i2s_write(I2S_NUM_0, (const char*)spkBuffer, sizeof(spkBuffer), &bytesRead, portMAX_DELAY);
}
#endif
