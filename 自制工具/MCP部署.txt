# main.py - 计算器 MCP 服务器（使用 FastMCP）
import math
from mcp.server.fastmcp import FastMCP

# 创建 FastMCP 实例
mcp = FastMCP("Calculator Server")

# 定义安全的 eval 环境
SAFE_NAMESPACE = {
    '__builtins__': None,
    'abs': abs,
    'round': round,
    'min': min,
    'max': max,
    'sum': sum,
    'pow': pow,
    'math': math,
    'sqrt': math.sqrt,
    'sin': math.sin,
    'cos': math.cos,
    'tan': math.tan,
    'log': math.log,
    'exp': math.exp,
    'pi': math.pi,
    'e': math.e,
}

@mcp.tool()
def calculate(expression: str) -> str:
    """
    计算数学表达式并返回结果。
    支持的运算：+ - * / ** % 以及 math 模块中的函数（如 sqrt, sin, cos 等）。
    示例：'1+2' -> 3, 'sqrt(16)' -> 4.0, '2**10' -> 1024
    """
    if not expression or not isinstance(expression, str):
        return "错误：请输入一个有效的表达式"

    try:
        code = compile(expression, '<string>', 'eval')
        result = eval(code, SAFE_NAMESPACE, {})
        return str(result)
    except SyntaxError:
        return "错误：表达式语法错误"
    except ZeroDivisionError:
        return "错误：除数不能为零"
    except Exception as e:
        return f"错误：{str(e)}"

if __name__ == "__main__":
    mcp.run(transport="stdio")

# mcp_client.py
import asyncio
import sys
import json
import os

class MCPClient:
    def __init__(self, server_script):
        self.server_script = server_script
        self.proc = None

    async def start(self):
        self.proc = await asyncio.create_subprocess_exec(
            sys.executable, self.server_script,
            stdin=asyncio.subprocess.PIPE,
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE
        )
        # 初始化
        await self._request({
            "jsonrpc": "2.0",
            "method": "initialize",
            "params": {
                "protocolVersion": "2025-11-25",
                "clientInfo": {"name": "custom-client", "version": "1.0"},
                "capabilities": {}
            },
            "id": 1
        })

    async def _request(self, req):
        self.proc.stdin.write((json.dumps(req) + "\n").encode())
        await self.proc.stdin.drain()
        resp = await self.proc.stdout.readline()
        return json.loads(resp.decode())

    async def call_tool(self, tool_name, arguments):
        req = {
            "jsonrpc": "2.0",
            "method": "tools/call",
            "params": {"name": tool_name, "arguments": arguments},
            "id": 2
        }
        resp = await self._request(req)
        return resp.get("result", {}).get("content", [])

    async def close(self):
        self.proc.terminate()
        await self.proc.wait()

async def main():
    client = MCPClient("main.py")
    await client.start()
    print("MCP 计算器客户端已启动。输入表达式如 '1+2'，输入 'exit' 退出。")
    while True:
        expr = input(">>> ")
        if expr.lower() in ("exit", "quit"):
            break
        result = await client.call_tool("calculate", {"expression": expr})
        # 提取文本结果
        for content in result:
            if content.get("type") == "text":
                print(content.get("text"))
    await client.close()

if __name__ == "__main__":
    asyncio.run(main())

# web_api.py
import asyncio
import json
import sys
from contextlib import asynccontextmanager
from typing import Any, Dict

import uvicorn
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel

# 请求体模型
class CalculateRequest(BaseModel):
    expression: str

# 响应体模型
class CalculateResponse(BaseModel):
    result: str

# 全局子进程和通信流
class MCPProcess:
    def __init__(self, script_path: str):
        self.script_path = script_path
        self.proc = None
        self.next_id = 1

    async def start(self):
        """启动 MCP 服务器子进程"""
        self.proc = await asyncio.create_subprocess_exec(
            sys.executable, self.script_path,
            stdin=asyncio.subprocess.PIPE,
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE
        )
        # 发送初始化请求（必须，MCP 协议要求）
        await self._request({
            "jsonrpc": "2.0",
            "method": "initialize",
            "params": {
                "protocolVersion": "2025-11-25",
                "clientInfo": {"name": "web-api-client", "version": "1.0"},
                "capabilities": {}
            }
        })

    async def _request(self, req: Dict[str, Any]) -> Dict[str, Any]:
        """发送 JSON-RPC 请求并等待响应"""
        req["id"] = self.next_id
        self.next_id += 1
        # 写入请求（每行一个 JSON）
        self.proc.stdin.write((json.dumps(req) + "\n").encode())
        await self.proc.stdin.drain()
        # 读取一行响应
        line = await self.proc.stdout.readline()
        if not line:
            # 可能进程已退出，读取 stderr 获取错误
            stderr = await self.proc.stderr.read()
            raise RuntimeError(f"MCP 进程异常退出: {stderr.decode()}")
        return json.loads(line.decode())

    async def call_tool(self, name: str, arguments: Dict[str, Any]) -> Any:
        """调用 MCP 工具"""
        req = {
            "jsonrpc": "2.0",
            "method": "tools/call",
            "params": {"name": name, "arguments": arguments}
        }
        resp = await self._request(req)
        if "error" in resp:
            raise RuntimeError(resp["error"].get("message", "未知错误"))
        # 提取结果内容
        result_content = resp["result"]["content"]
        # 假设第一个文本片段是结果
        for content in result_content:
            if content["type"] == "text":
                return content["text"]
        return str(result_content)

    async def stop(self):
        """终止子进程"""
        if self.proc:
            self.proc.terminate()
            await self.proc.wait()

# 全局 MCP 进程管理器
mcp_process = MCPProcess("main.py")

@asynccontextmanager
async def lifespan(app: FastAPI):
    # 启动时初始化 MCP 进程
    await mcp_process.start()
    yield
    # 关闭时清理
    await mcp_process.stop()

# 创建 FastAPI 应用
app = FastAPI(
    title="MCP Calculator Web API",
    description="通过 HTTP 调用 MCP 计算器服务器",
    lifespan=lifespan
)

@app.post("/calculate", response_model=CalculateResponse)

    """
    计算数学表达式
    - **expression**: 字符串形式的数学表达式，如 "1+2" 或 "sqrt(16)"
    """

        result = await mcp_process.call_tool("calculate", {"expression": req.expression})

@app.get("/health")
async def health():
    """健康检查端点"""
    return {"status": "ok"}

    uvicorn.run(app, host="0.0.0.0", port=8000)
