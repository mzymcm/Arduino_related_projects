// ==================== MAC 配置函数实现 ====================
void resetConfig() {
  config.magic = CONFIG_MAGIC;
  strcpy(config.ap_ssid, DEFAULT_AP_SSID);
  strcpy(config.ap_password, DEFAULT_AP_PASSWORD);
  memset(config.sta_ssid, 0, sizeof(config.sta_ssid));
  memset(config.sta_password, 0, sizeof(config.sta_password));
  memcpy(config.mac, DEFAULT_MAC, 6);
  config.sta_configured = false;
  config.ap_configured = true;
  saveConfig();
  WiFi.disconnect(true);
  WiFi.persistent(false);
}

void loadConfig() {
  EEPROM.get(0, config);
  if (config.magic != CONFIG_MAGIC || !config.ap_configured) {
    config.magic = CONFIG_MAGIC;
    strcpy(config.ap_ssid, "KeyAndMAC");
    strcpy(config.ap_password, "abcd1234");
    memset(config.sta_ssid, 0, sizeof(config.sta_ssid));
    memset(config.sta_password, 0, sizeof(config.sta_password));
    memset(config.mac, 0, sizeof(config.mac));
    config.sta_configured = false;
    config.ap_configured = true;
    saveConfig();
  }
}

void saveConfig() {
  EEPROM.put(0, config);
  EEPROM.commit();
}

void setAPMAC() {
  bool macIsZero = true;
  for (int i = 0; i < 6; i++) {
    if (config.mac[i] != 0x00) {
      macIsZero = false;
      break;
    }
  }
  if (!macIsZero) {
    wifi_set_macaddr(SOFTAP_IF, config.mac);
    delay(500);
  }
}

void restartAP() {
  WiFi.softAPdisconnect(true);
  delay(100);
  WiFi.softAP(config.ap_ssid, config.ap_password);
}

void connectToWiFi() {
  if (!config.sta_configured || strlen(config.sta_ssid) == 0) return;
  WiFi.begin(config.sta_ssid, config.sta_password);
  unsigned long startTime = millis();
  while (WiFi.status() != WL_CONNECTED && millis() - startTime < 20000) delay(500);
  sta_connected = (WiFi.status() == WL_CONNECTED);
  if (!sta_connected) WiFi.disconnect();
}

String macToString(uint8_t* mac) {
  char macStr[18];
  snprintf(macStr, 18, "%02X:%02X:%02X:%02X:%02X:%02X", mac[0], mac[1], mac[2], mac[3], mac[4], mac[5]);
  return String(macStr);
}

String bssidToString(uint8_t* bssid) {
  char bssidStr[18];
  snprintf(bssidStr, 18, "%02X:%02X:%02X:%02X:%02X:%02X", bssid[0], bssid[1], bssid[2], bssid[3], bssid[4], bssid[5]);
  return String(bssidStr);
}

bool parseMAC(const String& macStr, uint8_t* mac) {
  String clean = macStr;
  clean.replace(":", "");
  clean.replace(" ", "");
  clean.replace("-", "");
  clean.toUpperCase();
  if (clean.length() != 12) return false;
  for (int i = 0; i < 12; i++) {
    char c = clean[i];
    if (!((c >= '0' && c <= '9') || (c >= 'A' && c <= 'F'))) return false;
  }
  for (int i = 0; i < 6; i++) {
    String byteStr = clean.substring(i*2, i*2+2);
    mac[i] = strtol(byteStr.c_str(), NULL, 16);
  }
  return true;
}

String getEncryptionTypeStr(uint8_t encryptionType) {
  switch(encryptionType) {
    case ENC_TYPE_NONE: return "开放网络";
    case ENC_TYPE_WEP: return "WEP";
    case ENC_TYPE_TKIP: return "WPA";
    case ENC_TYPE_CCMP: return "WPA2";
    default: return "未知";
  }
}

void sortNetworksByRSSI(int* indices, int n) {
  for (int i = 0; i < n-1; i++)
    for (int j = i+1; j < n; j++)
      if (WiFi.RSSI(indices[j]) > WiFi.RSSI(indices[i])) {
        int t = indices[i]; indices[i] = indices[j]; indices[j] = t;
      }
}

String getSignalStrengthClass(int rssi) {
  if (rssi >= -50) return "signal-strong";
  else if (rssi >= -70) return "signal-medium";
  else return "signal-weak";
}

}

void screenToChip(int screenX, int screenY, uint8_t &lowX, uint8_t &highX, uint8_t &lowY, uint8_t &highY) {
  uint16_t chipX = (uint32_t)screenX * 4096 / screenWidth;
  uint16_t chipY = (uint32_t)screenY * 4096 / screenHeight;
  lowX = chipX & 0xFF;
  highX = (chipX >> 8) & 0xFF;
  lowY = chipY & 0xFF;
  highY = (chipY >> 8) & 0xFF;
}

void sendAbsoluteMouse(uint8_t buttons, int screenX, int screenY, uint8_t wheel) {
  if (screenX < 0) screenX = 0;
  if (screenX > screenWidth) screenX = screenWidth;
  if (screenY < 0) screenY = 0;
  if (screenY > screenHeight) screenY = screenHeight;
  uint8_t lowX, highX, lowY, highY;
  screenToChip(screenX, screenY, lowX, highX, lowY, highY);
  uint8_t packet[13];
  packet[0] = 0x57;
  packet[1] = 0xAB;
  packet[2] = 0x00;
  packet[3] = 0x04;
  packet[4] = 0x07;
  packet[5] = 0x02;
  packet[6] = buttons;
  packet[7] = lowX;
  packet[8] = highX;
  packet[9] = lowY;
  packet[10] = highY;
  packet[11] = wheel;
  uint8_t sum = 0;
  for (int i = 0; i < 12; i++) sum += packet[i];
  packet[12] = sum;
  CH9329_SERIAL.write(packet, 13);
  delay(2);
}

void sendChar(char ch) {
  uint8_t modifier = 0x00, keycode = 0x00;
  if (ch >= 'a' && ch <= 'z') {
    keycode = ch - 'a' + KEY_A;
  } else if (ch >= 'A' && ch <= 'Z') {
    modifier = MOD_LEFT_SHIFT;
    keycode = ch - 'A' + KEY_A;
  } else if (ch >= '0' && ch <= '9') {
    keycode = (ch == '0') ? KEY_0 : KEY_1 + (ch - '1');
  } else {
    switch(ch) {
      case ' ': keycode = KEY_SPACE; break;
      case '\n': case '\r': keycode = KEY_ENTER; break;
      case '\t': keycode = KEY_TAB; break;
      case '-': keycode = KEY_MINUS; break;
      case '=': keycode = KEY_EQUAL; break;
      case '[': keycode = KEY_LEFT_BRACKET; break;
      case ']': keycode = KEY_RIGHT_BRACKET; break;
      case '\\': keycode = KEY_BACKSLASH; break;
      case ';': keycode = KEY_SEMICOLON; break;
      case '\'': keycode = KEY_QUOTE; break;
      case '`': keycode = KEY_GRAVE; break;
      case ',': keycode = KEY_COMMA; break;
      case '.': keycode = KEY_PERIOD; break;
      case '/': keycode = KEY_SLASH; break;
      case '!': modifier = MOD_LEFT_SHIFT; keycode = KEY_1; break;
      case '@': modifier = MOD_LEFT_SHIFT; keycode = KEY_2; break;
      case '#': modifier = MOD_LEFT_SHIFT; keycode = KEY_3; break;
      case '$': modifier = MOD_LEFT_SHIFT; keycode = KEY_4; break;
      case '%': modifier = MOD_LEFT_SHIFT; keycode = KEY_5; break;
      case '^': modifier = MOD_LEFT_SHIFT; keycode = KEY_6; break;
      case '&': modifier = MOD_LEFT_SHIFT; keycode = KEY_7; break;
      case '*': modifier = MOD_LEFT_SHIFT; keycode = KEY_8; break;
      case '(': modifier = MOD_LEFT_SHIFT; keycode = KEY_9; break;
      case ')': modifier = MOD_LEFT_SHIFT; keycode = KEY_0; break;
      case '_': modifier = MOD_LEFT_SHIFT; keycode = KEY_MINUS; break;
      case '+': modifier = MOD_LEFT_SHIFT; keycode = KEY_EQUAL; break;
      case '{': modifier = MOD_LEFT_SHIFT; keycode = KEY_LEFT_BRACKET; break;
      case '}': modifier = MOD_LEFT_SHIFT; keycode = KEY_RIGHT_BRACKET; break;
      case '|': modifier = MOD_LEFT_SHIFT; keycode = KEY_BACKSLASH; break;
      case ':': modifier = MOD_LEFT_SHIFT; keycode = KEY_SEMICOLON; break;
      case '"': modifier = MOD_LEFT_SHIFT; keycode = KEY_QUOTE; break;
      case '~': modifier = MOD_LEFT_SHIFT; keycode = KEY_GRAVE; break;
      case '<': modifier = MOD_LEFT_SHIFT; keycode = KEY_COMMA; break;
      case '>': modifier = MOD_LEFT_SHIFT; keycode = KEY_PERIOD; break;
      case '?': modifier = MOD_LEFT_SHIFT; keycode = KEY_SLASH; break;
      default: return;
    }
  }
  if (keycode) sendSingleKey(modifier, keycode);
}

void sendUnicodeChar_Hex(uint32_t codepoint) {
  char hex[9];
  snprintf(hex, sizeof(hex), "%X", codepoint);
  sendKeyboardReport(MOD_LEFT_ALT, 0x00);
  delay(50);
  sendKeyboardReport(MOD_LEFT_ALT, KEY_KP_PLUS);
  sendKeyboardReport(MOD_LEFT_ALT, 0x00);
  delay(30);
  for (char* p = hex; *p; p++) {
    char c = *p;
    uint8_t modifier = MOD_LEFT_ALT;
    uint8_t keycode = 0;
    if (c >= '0' && c <= '9') {
      switch(c) {
        case '0': keycode = KEY_KP_0; break;
        case '1': keycode = KEY_KP_1; break;
        case '2': keycode = KEY_KP_2; break;
        case '3': keycode = KEY_KP_3; break;
        case '4': keycode = KEY_KP_4; break;
        case '5': keycode = KEY_KP_5; break;
        case '6': keycode = KEY_KP_6; break;
        case '7': keycode = KEY_KP_7; break;
        case '8': keycode = KEY_KP_8; break;
        case '9': keycode = KEY_KP_9; break;
      }
    } else if (c >= 'A' && c <= 'F') {
      modifier |= MOD_LEFT_SHIFT;
      keycode = KEY_A + (c - 'A');
    } else continue;
    sendKeyboardReport(modifier, keycode);
    sendKeyboardReport(modifier, 0x00);
    delay(20);
  }
  sendKeyboardReport(0, 0x00);
  delay(40);
}

void sendUnicodeChar_LocationCode(uint32_t codepoint) {
  if (codepoint < 0x4E00 || codepoint > 0x9FA5) {
    sendUnicodeChar_Hex(codepoint);
    return;
  }
  uint16_t offset = codepoint - 0x4E00;
  uint8_t qu = offset / 94 + 16;
  uint8_t wei = offset % 94 + 1;
  char locCode[5];
  sprintf(locCode, "%02d%02d", qu, wei);
  sendKeyboardReport(MOD_LEFT_CTRL, 0x00);
  sendKeyboardReport(MOD_LEFT_CTRL, KEY_LEFT_BRACKET);
  sendKeyboardReport(MOD_LEFT_CTRL, 0x00);
  sendKeyboardReport(0x00, 0x00);
  delay(80);
  for (char* p = locCode; *p; p++) {
    uint8_t keycode = KEY_1 + (*p - '1');
    if (*p == '0') keycode = KEY_0;
    sendSingleKey(0x00, keycode);
    delay(20);
  }
  sendSingleKey(0x00, KEY_ENTER);
  delay(80);
}

void sendText(const String& text) {
  if (chineseInputMethod == 0) {
    for (size_t i = 0; i < text.length(); i++) {
      char ch = text[i];
      if (ch >= 0x20 && ch <= 0x7E) sendChar(ch);
      else if (ch == '\n') sendChar('\n');
      delay(15);
    }
    return;
  }
  for (size_t i = 0; i < text.length(); i++) {
    char ch = text[i];
    if (ch & 0x80) {
      uint32_t codepoint = 0;
      int bytes = 0;
      if ((ch & 0xE0) == 0xC0) { bytes = 2; codepoint = ch & 0x1F; }
      else if ((ch & 0xF0) == 0xE0) { bytes = 3; codepoint = ch & 0x0F; }
      else if ((ch & 0xF8) == 0xF0) { bytes = 4; codepoint = ch & 0x07; }
      else continue;
      for (int j = 1; j < bytes && i+j < text.length(); j++) {
        uint8_t next = text[i+j];
        if ((next & 0xC0) != 0x80) break;
        codepoint = (codepoint << 6) | (next & 0x3F);
      }
      if (chineseInputMethod == 1) sendUnicodeChar_Hex(codepoint);
      else if (chineseInputMethod == 2) sendUnicodeChar_LocationCode(codepoint);
      else if (chineseInputMethod == 3) sendUnicodeChar_Hex(codepoint);
      i += bytes - 1;
    } else {
      sendChar(ch);
    }
    delay(20);
  }
}

void handleSave() {
  String type = server.arg("type");
  if (type == "ap") {
    String ap_ssid = server.arg("ap_ssid");
    String ap_password = server.arg("ap_password");
    if (ap_ssid.length() > 0 && ap_ssid.length() <= 31) {
      ap_ssid.toCharArray(config.ap_ssid, sizeof(config.ap_ssid));
      ap_password.toCharArray(config.ap_password, sizeof(config.ap_password));
      saveConfig();
      restartAP();
      server.send(200, "text/html; charset=utf-8", "<script>alert('AP设置已保存，热点已重启');window.location='/mac';</script>");
    } else {
      server.send(200, "text/html; charset=utf-8", "<script>alert('AP名称长度必须在1-31个字符之间');window.location='/mac';</script>");
    }
  }
  else if (type == "wifi") {
    String sta_ssid = server.arg("sta_ssid");
    String sta_password = server.arg("sta_password");
    if (sta_ssid.length() > 0 && sta_ssid.length() <= 31) {
      sta_ssid.toCharArray(config.sta_ssid, sizeof(config.sta_ssid));
      sta_password.toCharArray(config.sta_password, sizeof(config.sta_password));
      config.sta_configured = true;
      saveConfig();
      connectToWiFi();
      server.send(200, "text/html; charset=utf-8", "<script>alert('WiFi设置已保存');window.location='/mac';</script>");
    } else {
      server.send(200, "text/html; charset=utf-8", "<script>alert('WiFi名称长度必须在1-31个字符之间');window.location='/mac';</script>");
    }
  }
  else if (type == "mac") {
    String macStr = server.arg("mac_address");
    macStr.trim();
    if (macStr.length() == 0) {
      memcpy(config.mac, DEFAULT_MAC, 6);
      saveConfig();
      setAPMAC();
      server.send(200, "text/html; charset=utf-8", "<script>alert('已恢复默认MAC地址');window.location='/mac';</script>");
    } else if (parseMAC(macStr, config.mac)) {
      saveConfig();
      setAPMAC();
      server.send(200, "text/html; charset=utf-8", "<script>alert('MAC地址已保存');window.location='/mac';</script>");
    } else {
      server.send(200, "text/html; charset=utf-8", "<script>alert('MAC地址格式错误，请使用格式: 06:69:6C:45:CA:21');window.location='/mac';</script>");
    }
  }
}

void handleSpecial() {
  if (!server.hasArg("key")) { server.send(400, "text/plain", "Missing key"); return; }
  String key = server.arg("key");
  // 原有特殊键
  if (key == "enter") sendSingleKey(0x00, KEY_ENTER);
  else if (key == "backspace") sendSingleKey(0x00, KEY_BACKSPACE);
  else if (key == "tab") sendSingleKey(0x00, KEY_TAB);
  else if (key == "backslash") sendSingleKey(0x00, KEY_BACKSLASH);
  else if (key == "shift") sendSingleKey(MOD_LEFT_SHIFT, 0x00);
  else if (key == "ctrl") sendSingleKey(MOD_LEFT_CTRL, 0x00);
  else if (key == "ctrlspace") sendSingleKey(MOD_LEFT_CTRL, KEY_SPACE);
  // 方向键
  else if (key == "up") sendSingleKey(0x00, KEY_UP);
  else if (key == "down") sendSingleKey(0x00, KEY_DOWN);
  else if (key == "left") sendSingleKey(0x00, KEY_LEFT);
  else if (key == "right") sendSingleKey(0x00, KEY_RIGHT);
  // 组合键
  else if (key == "ctrl_c") sendCombination(MOD_LEFT_CTRL, KEY_C);
  else if (key == "ctrl_v") sendCombination(MOD_LEFT_CTRL, KEY_V);
  else if (key == "ctrl_x") sendCombination(MOD_LEFT_CTRL, KEY_X);
  else if (key == "ctrl_z") sendCombination(MOD_LEFT_CTRL, KEY_Z);
  else if (key == "ctrl_a") sendCombination(MOD_LEFT_CTRL, KEY_A);
  else if (key == "ctrl_alt_del") sendCombination(MOD_LEFT_CTRL | MOD_LEFT_ALT, KEY_DELETE);
  else if (key == "ctrl_shift_esc") sendCombination(MOD_LEFT_CTRL | MOD_LEFT_SHIFT, KEY_ESC);
  else if (key == "alt_tab") sendCombination(MOD_LEFT_ALT, KEY_TAB);
  else if (key == "win_r") sendCombination(MOD_LEFT_GUI, KEY_R);
  else if (key == "win_e") sendCombination(MOD_LEFT_GUI, KEY_E);
  else if (key == "win_d") sendCombination(MOD_LEFT_GUI, KEY_D);
  else { server.send(400, "text/plain", "Unknown special"); return; }
  server.send(200, "text/plain", "OK");
}

// ==================== 主程序 ====================
void setup() {
  Serial.begin(9600);
  delay(200);
  EEPROM.begin(EEPROM_SIZE);
  loadConfig();
  WiFi.disconnect(true);
  WiFi.persistent(false);
  delay(100);
  WiFi.mode(WIFI_AP_STA);
  delay(100);
  setAPMAC();
  IPAddress local_IP(192, 168, 4, 1);
  IPAddress gateway(192, 168, 4, 1);
  IPAddress subnet(255, 255, 255, 0);
  WiFi.softAPConfig(local_IP, gateway, subnet);
  WiFi.softAP(config.ap_ssid, config.ap_password, 6);
  connectToWiFi();
  server.on("/", handleRoot);
  server.on("/mac", handleMacPage);
  server.on("/save", HTTP_POST, handleSave);
  server.on("/scan", handleScan);
  server.on("/scanmac", handleScanMAC);
  server.on("/status", handleStatus);
  server.on("/reset", handleReset);
  server.on("/reboot", handleReboot);
  server.on("/hid", handleHidPage);
  server.on("/key", handleKey);
  server.on("/special", handleSpecial);
  server.on("/mouse_move", handleMouseMove);
  server.on("/mouse_move_locked", handleMouseMoveLocked);
  server.on("/mouse_press_left", handleMousePressLeft);
  server.on("/mouse_release_left", handleMouseReleaseLeft);
  server.on("/mouse_click", handleMouseClick);
  server.on("/set_res", handleSetRes);
  server.on("/text", HTTP_POST, handleText);
  server.onNotFound(handleNotFound);
  server.begin();
}

void loop() {
  server.handleClient();
  unsigned long currentMillis = millis();
  if (currentMillis - previousMillis >= interval) {
    previousMillis = currentMillis;
    if (config.sta_configured) checkWiFiStatus();
  }
  delay(1);
}
