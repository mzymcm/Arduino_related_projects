/*
 * ESP8266 (ESP01) + CH9329 协议模式 - 网页虚拟键盘（单键发送 + Ctrl+空格）
 * 硬件：CH9329 连接至 UART0 (GPIO1=TX, GPIO3=RX)
 * 模式：协议传输模式（默认），波特率9600
 * 功能：ESP8266 开启 WiFi 热点，浏览器访问 192.168.4.1，点击按键即发送对应键盘敲击
 *      新增 Ctrl+空格 组合键，用于关闭/切换中文输入法
 * CFG0/CFG1/MODE0/MODE1 全部悬空（默认模式0，协议传输）
 * 
 * 注意：烧录时需断开 CH9329 与 GPIO1/GPIO3 的连接，烧录完成后再接上。
 */

#define KEY_COMMA   0x36
#define KEY_PERIOD  0x37
#define KEY_SLASH   0x38

#define MOD_LEFT_SHIFT  0x02
#define MOD_LEFT_CTRL   0x01   // 新增左Ctrl修饰符

// ========== 发送单个按键（按下+释放） ==========
void sendSingleKey(uint8_t modifier, uint8_t keycode) {
  uint8_t packet[14];
  packet[0] = 0x57;
  packet[1] = 0xAB;
  packet[2] = 0x00;
  packet[3] = 0x02;
  packet[4] = 0x08;
  packet[5] = modifier;
  packet[6] = 0x00;
  packet[7] = keycode;
  packet[8] = 0x00;
  packet[9] = 0x00;
  packet[10] = 0x00;
  packet[11] = 0x00;
  packet[12] = 0x00;
  uint8_t sum = 0;
  for (int i = 0; i < 13; i++) sum += packet[i];
  packet[13] = sum;
  
  CH9329_SERIAL.write(packet, 14);
  delay(20);
  
  // 释放按键包（全0）
  uint8_t release[14] = {0x57, 0xAB, 0x00, 0x02, 0x08, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x0C};
  CH9329_SERIAL.write(release, 14);
  delay(10);
}

// 发送字符（支持大小写、数字、常用符号）
void sendChar(char ch) {
  uint8_t modifier = 0x00;
  uint8_t keycode = 0x00;
  
  if (ch >= 'a' && ch <= 'z') {
    keycode = ch - 'a' + KEY_A;
  } else if (ch >= 'A' && ch <= 'Z') {
    modifier = MOD_LEFT_SHIFT;
    keycode = ch - 'A' + KEY_A;
  } else if (ch >= '0' && ch <= '9') {
    if (ch == '0') keycode = KEY_0;
    else keycode = KEY_1 + (ch - '1');
  } else {
    switch(ch) {
      case ' ': keycode = KEY_SPACE; break;
      case '\n': keycode = KEY_ENTER; break;
      case '\r': keycode = KEY_ENTER; break;
      case '\t': keycode = KEY_TAB; break;
      case '-': keycode = KEY_MINUS; break;
      case '=': keycode = KEY_EQUAL; break;
      case '[': keycode = KEY_LEFT_BRACKET; break;
      case ']': keycode = KEY_RIGHT_BRACKET; break;
      case '\\': keycode = KEY_BACKSLASH; break;
      case ';': keycode = KEY_SEMICOLON; break;
      case '\'': keycode = KEY_QUOTE; break;
      case '`': keycode = KEY_GRAVE; break;
      case ',': keycode = KEY_COMMA; break;
      case '.': keycode = KEY_PERIOD; break;
      case '/': keycode = KEY_SLASH; break;
      case '!': modifier = MOD_LEFT_SHIFT; keycode = KEY_1; break;
      case '@': modifier = MOD_LEFT_SHIFT; keycode = KEY_2; break;
      case '#': modifier = MOD_LEFT_SHIFT; keycode = KEY_3; break;
      case '$': modifier = MOD_LEFT_SHIFT; keycode = KEY_4; break;
      case '%': modifier = MOD_LEFT_SHIFT; keycode = KEY_5; break;
      case '^': modifier = MOD_LEFT_SHIFT; keycode = KEY_6; break;
      case '&': modifier = MOD_LEFT_SHIFT; keycode = KEY_7; break;
      case '*': modifier = MOD_LEFT_SHIFT; keycode = KEY_8; break;
      case '(': modifier = MOD_LEFT_SHIFT; keycode = KEY_9; break;
      case ')': modifier = MOD_LEFT_SHIFT; keycode = KEY_0; break;
      case '_': modifier = MOD_LEFT_SHIFT; keycode = KEY_MINUS; break;
      case '+': modifier = MOD_LEFT_SHIFT; keycode = KEY_EQUAL; break;
      case '{': modifier = MOD_LEFT_SHIFT; keycode = KEY_LEFT_BRACKET; break;
      case '}': modifier = MOD_LEFT_SHIFT; keycode = KEY_RIGHT_BRACKET; break;
      case '|': modifier = MOD_LEFT_SHIFT; keycode = KEY_BACKSLASH; break;
      case ':': modifier = MOD_LEFT_SHIFT; keycode = KEY_SEMICOLON; break;
      case '"': modifier = MOD_LEFT_SHIFT; keycode = KEY_QUOTE; break;
      case '~': modifier = MOD_LEFT_SHIFT; keycode = KEY_GRAVE; break;
      case '<': modifier = MOD_LEFT_SHIFT; keycode = KEY_COMMA; break;
      case '>': modifier = MOD_LEFT_SHIFT; keycode = KEY_PERIOD; break;
      case '?': modifier = MOD_LEFT_SHIFT; keycode = KEY_SLASH; break;
      default: return;
    }
  }
  if (keycode != 0) {
    sendSingleKey(modifier, keycode);
  }
}

// 发送特殊功能键
void sendSpecialKey(uint8_t keycode) {
  sendSingleKey(0x00, keycode);
}

// ========== Web 路由 ==========
void handleRoot() {
  server.send(200, "text/html", index_html);
}

void handleKey() {
  if (server.hasArg("char")) {
    String ch = server.arg("char");
    if (ch.length() == 1) {
      sendChar(ch[0]);
      server.send(200, "text/plain", "OK");
    } else {
      server.send(400, "text/plain", "Invalid char");
    }
  } else {
    server.send(400, "text/plain", "Missing char");
  }
}

void handleSpecial() {
  if (server.hasArg("key")) {
    String key = server.arg("key");
    if (key == "enter") {
      sendSpecialKey(KEY_ENTER);
    } else if (key == "backspace") {
      sendSpecialKey(KEY_BACKSPACE);
    } else if (key == "tab") {
      sendSpecialKey(KEY_TAB);
    } else if (key == "backslash") {
      sendSpecialKey(KEY_BACKSLASH);
    } else if (key == "shift") {
      sendSingleKey(MOD_LEFT_SHIFT, 0x00);
    } else if (key == "ctrl") {
      sendSingleKey(MOD_LEFT_CTRL, 0x00);      // 左 Ctrl 单独按下释放
    } else if (key == "ctrlspace") {            // 新增：Ctrl+空格 组合键
      sendSingleKey(MOD_LEFT_CTRL, KEY_SPACE);  // 同时按下 Ctrl 和 空格
    } else {
      server.send(400, "text/plain", "Unknown special");
      return;
    }
    server.send(200, "text/plain", "OK");
  } else {
    server.send(400, "text/plain", "Missing key");
  }
}

void handleNotFound() {
  server.send(404, "text/plain", "404");
}

// ========== Setup ==========
void setup() {
  // 初始化硬件串口与 CH9329 通信，波特率 9600
  CH9329_SERIAL.begin(9600);
  delay(100);

  // 配置 WiFi 为 AP 模式（无调试输出，避免干扰串口）
  WiFi.mode(WIFI_AP);
  WiFi.softAP(ap_ssid, ap_password);
  // IP 地址固定为 192.168.4.1，无需打印

  // 启动 Web 服务器
  server.on("/", handleRoot);
  server.on("/key", handleKey);
  server.on("/special", handleSpecial);
  server.onNotFound(handleNotFound);
  server.begin();
}

void loop() {
  server.handleClient();
  delay(5);
}
