将以下代码复制到一个新文件中。它演示了如何初始化屏幕、点亮背光并显示一些系统信息（如时间、CPU温度等）。你可以根据需要修改脚本。


#!/usr/bin/env python3
import time
import subprocess
from PIL import Image, ImageDraw, ImageFont
from luma.core.interface.serial import spi
from luma.lcd.device import st7735

# 初始化屏幕，指定DC和RST引脚
serial = spi(port=0, device=0, gpio_DC=25, gpio_RST=27)
device = st7735(serial, rotate=0, width=128, height=128)

# 初始化背光GPIO
import RPi.GPIO as GPIO
GPIO.setmode(GPIO.BCM)
GPIO.setup(23, GPIO.OUT)
GPIO.output(23, GPIO.HIGH)

def get_cpu_temp():
    try:
        with open("/sys/class/thermal/thermal_zone0/temp", "r") as f:
            temp = float(f.read()) / 1000
            return f"{temp:.1f}°C"
    except:
        return "N/A"

def get_ip():
    try:
        result = subprocess.run(['hostname', '-I'], capture_output=True, text=True)
        return result.stdout.split()[0]
    except:
        return "No IP"

font = ImageFont.load_default()
while True:
    with Image.new("RGB", device.size) as img:
        draw = ImageDraw.Draw(img)
        draw.rectangle(device.bounding_box, fill="black")
        draw.text((5, 5), f"Time: {time.strftime('%H:%M:%S')}", fill="white", font=font)
        draw.text((5, 25), f"CPU Temp: {get_cpu_temp()}", fill="white", font=font)
        draw.text((5, 45), f"IP: {get_ip()}", fill="white", font=font)
        draw.text((5, 65), "RPi 1B Online", fill="white", font=font)
        device.display(img)
    time.sleep(2)

使用 fbtft 驱动，直接指定控制器 st7789v，显式设置旋转后的尺寸、BGR 和反转模式，解决了所有颜色、杂条问题。现在只需把终端映射和开机自启设置好，就能投入日常使用。

最终配置（保留在 /boot/firmware/config.txt）

dtparam=spi=on
dtoverlay=fbtft,spi0-0,st7789v,dc_pin=25,reset_pin=27,led_pin=23,speed=24000000,rotate=90,width=240,height=320,buswidth=8,bgr,invert,fps=30
说明：

spi0-0 → 使用 SPI0、片选 CE0（你的接线）

st7789v → 控制器名，使用驱动内置初始化

dc_pin、reset_pin、led_pin → 匹配你的 GPIO

speed=24000000 → 24 MHz，稳定；可尝试 32000000 提升刷新率

rotate=90 → 竖屏方向（可根据需要改为 0/180/270）

width=240,height=320 → 旋转后的正确尺寸，消除杂边

bgr,invert → 修正红蓝互换和黑白颠倒

buswidth=8 → 确保数据宽度正确

fps=30 → 每秒最大帧数

// ---------- 底层写函数（支持硬件/软件SPI） ----------
static void sw_spi_write(uint8_t data) {
    for (int i = 7; i >= 0; i--) {
        digitalWrite(PIN_SCK, LOW);
        digitalWrite(PIN_MOSI, (data >> i) & 1);
        digitalWrite(PIN_SCK, HIGH);
    }
}

static void st7735_write_cmd(uint8_t cmd) {
    digitalWrite(PIN_DC, LOW);
#if USE_HARDWARE_SPI
    wiringPiSPIDataRW(0, &cmd, 1);
#else
    sw_spi_write(cmd);
#endif
}

static void st7735_write_data(uint8_t data) {
    digitalWrite(PIN_DC, HIGH);
#if USE_HARDWARE_SPI
    wiringPiSPIDataRW(0, &data, 1);
#else
    sw_spi_write(data);
#endif
}

static void st7735_write_data_array(uint8_t *data, int len) {
    digitalWrite(PIN_DC, HIGH);
#if USE_HARDWARE_SPI
    wiringPiSPIDataRW(0, data, len);
#else
    for (int i = 0; i < len; i++) sw_spi_write(data[i]);
#endif
}

// ---------- 绘制彩色条纹（测试用） ----------
static void draw_stripes(void) {
    uint16_t *buffer = (uint16_t *)malloc(WIDTH * HEIGHT * sizeof(uint16_t));
    if (!buffer) return;
    int stripe_width = WIDTH / 5;
    uint16_t colors[5] = {0x0000, 0xFFFF, 0xF800, 0x07E0, 0x001F}; // 黑、白、红、绿、蓝
    for (int y = 0; y < HEIGHT; y++) {
        for (int x = 0; x < WIDTH; x++) {
            int idx = x / stripe_width;
            if (idx >= 5) idx = 4;
            buffer[y * WIDTH + x] = colors[idx];
        }
    }
    st7735_write_cmd(ST7735_RAMWR);
    for (int i = 0; i < WIDTH * HEIGHT; i++) {
        uint8_t bytes[2] = {buffer[i] >> 8, buffer[i] & 0xFF};
        st7735_write_data_array(bytes, 2);
    }
    free(buffer);
}

// ---------- JPEG 解码与转换 ----------
static unsigned char *read_jpeg(const char *filename, int *width, int *height) {
    FILE *file = fopen(filename, "rb");
    if (!file) {
        perror("打开JPEG失败");
        return NULL;
    }
    struct jpeg_decompress_struct cinfo;
    struct jpeg_error_mgr jerr;
    cinfo.err = jpeg_std_error(&jerr);
    jpeg_create_decompress(&cinfo);
    jpeg_stdio_src(&cinfo, file);
    jpeg_read_header(&cinfo, TRUE);
    jpeg_start_decompress(&cinfo);

    *width = cinfo.output_width;
    *height = cinfo.output_height;
    int row_stride = cinfo.output_width * cinfo.output_components;
    unsigned char *buffer = (unsigned char *)malloc(row_stride * cinfo.output_height);
    if (!buffer) {
        jpeg_finish_decompress(&cinfo);
        jpeg_destroy_decompress(&cinfo);
        fclose(file);
        return NULL;
    }
    unsigned char *row_ptr = buffer;
    while (cinfo.output_scanline < cinfo.output_height) {
        jpeg_read_scanlines(&cinfo, &row_ptr, 1);
        row_ptr += row_stride;
    }
    jpeg_finish_decompress(&cinfo);
    jpeg_destroy_decompress(&cinfo);
    fclose(file);
    return buffer;
}

static uint16_t *jpeg_to_rgb565(const char *filename) {
    int img_w, img_h;
    unsigned char *rgb24 = read_jpeg(filename, &img_w, &img_h);
    if (!rgb24) return NULL;

    int w = (img_w < WIDTH) ? img_w : WIDTH;
    int h = (img_h < HEIGHT) ? img_h : HEIGHT;

    uint16_t *rgb565 = (uint16_t *)malloc(WIDTH * HEIGHT * sizeof(uint16_t));
    if (!rgb565) { free(rgb24); return NULL; }
    memset(rgb565, 0, WIDTH * HEIGHT * sizeof(uint16_t));

    for (int y = 0; y < h; y++) {
        unsigned char *row = rgb24 + y * img_w * 3;
        uint16_t *dest = rgb565 + y * WIDTH;
        for (int x = 0; x < w; x++) {
            unsigned char r = row[x*3], g = row[x*3+1], b = row[x*3+2];
            dest[x] = ((r >> 3) << 11) | ((g >> 2) << 5) | (b >> 3);
        }
    }
    free(rgb24);
    return rgb565;
}

    }
}
