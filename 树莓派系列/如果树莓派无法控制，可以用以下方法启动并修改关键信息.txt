1. 安装与配置

【....】问题和解决：
网络依赖导致的服务启动超时
pigpiod 服务在启动时，可能需要等待网络目标（network.target）就绪。在没有插入 USB 网卡的情况下，系统可能会等待一个根本不存在的 eth0 等网络接口，导致启动超时或失败。
在极少数网络配置混乱的情况下，localhost 的解析可能会出问题。正常情况下，/etc/hosts 文件中应该包含将 localhost 指向 127.0.0.1 和 ::1 的条目。你可以通过 cat /etc/hosts 命令来快速检查一下。
# 1. 查找并强制结束所有 pigpiod 进程
sudo killall pigpiod

# 2. 检查 8888 端口是否已被释放
sudo netstat -tlnp | grep 8888

# 3. 确认无输出后，重新启动服务
sudo systemctl start pigpiod

正常情况下，/etc/hosts 必须包含至少下面这两行：
127.0.0.1       localhost
::1             localhost ip6-localhost ip6-loopback
你的文件里完全没有 127.0.0.1 localhost 这一条，也没有 IPv6 的 loopback 定义。这导致系统无法解析 localhost 这个主机名，当 pigpiod 尝试绑定到 localhost（默认行为）或任何代码尝试连接 localhost:8888 时，都会因为主机名无法解析而报错。

127.0.0.1 localhost
127.0.1.1 DietPi
::1 localhost ip6-localhost ip6-loopback
ff02::1 ip6-allnodes
ff02::2 ip6-allrouters

#!/usr/bin/env python3
"""
ST7789 视频播放控制器
"""

import pigpio
import threading
import time
import sys
import glob
import re

# ========== 配置 ==========
TX_PIN = 19
RX_PIN = 16
BAUD = 38400
BITS = 8

def get_last_framebuffer():
    fb_devices = sorted(glob.glob('/dev/fb[0-9]*'))
    return fb_devices[-1] if fb_devices else None

FB_DEVICE = get_last_framebuffer()
if FB_DEVICE is None:
    print("❌ 未找到任何帧缓冲设备 (/dev/fb*)")
    sys.exit(1)
print(f"✅ 使用帧缓冲设备: {FB_DEVICE}")

VIDEO_FILE = "/root/themy/AV/test_320x240_inv.rgb"
SCREEN_WIDTH = 320
SCREEN_HEIGHT = 240
BYTES_PER_PIXEL = 2
FRAME_SIZE = SCREEN_WIDTH * SCREEN_HEIGHT * BYTES_PER_PIXEL
FPS = 30

LED_PIN = 18
LED_ON = 1

EXIT_CONFIRM_TIMEOUT = 2.0   # 两次 exit 的最大间隔
CMD_COOLDOWN = 0.3           # 同类指令冷却时间（秒）

# ========== 连接 pigpiod（带重试） ==========
def connect_pigpio(host='127.0.0.1', port=8888, retries=20, delay=0.5):
    for i in range(retries):
        pi = pigpio.pi(host, port)
        if pi.connected:
            print(f"✅ 已连接到 pigpiod ({host}:{port})")
            return pi
        pi.stop()
        print(f"⏳ 等待 pigpiod 就绪... ({i+1}/{retries})")
        time.sleep(delay)
    print("❌ 无法连接 pigpiod，请检查服务状态")
    sys.exit(1)

pi = connect_pigpio('127.0.0.1', 8888)

running = True
video_playing = False
playback_thread = None
video_stop_event = threading.Event()

exit_pending_time = None      # 用于 exit 二次确认
last_play_time = 0
last_stop_time = 0

# 读取视频文件
try:
    with open(VIDEO_FILE, "rb") as f:
        video_data = f.read()
except FileNotFoundError:
    print(f"❌ 视频文件未找到: {VIDEO_FILE}")
    pi.stop()
    sys.exit(1)

video_size = len(video_data)
if video_size % FRAME_SIZE == 0:
    total_frames = video_size // FRAME_SIZE
    print(f"✅ 检测到 {total_frames} 帧视频")
else:
    total_frames = 1
    print(f"⚠️ 文件大小异常，将作为单帧循环播放")

# ========== 显示控制 ==========
def clear_screen():
    try:
        with open(FB_DEVICE, 'wb') as fb:
            fb.write(b'\x00' * FRAME_SIZE)
    except Exception as e:
        print(f"⚠️ 清屏失败: {e}")

def set_backlight(on: bool):
    pi.write(LED_PIN, LED_ON if on else (1 - LED_ON))

def play_video_loop():
    global video_playing
    print("▶️  开始播放视频")
    set_backlight(True)
    frame_index = 0
    frame_duration = 1.0 / FPS
    while not video_stop_event.is_set():
        if total_frames > 1:
            offset = (frame_index % total_frames) * FRAME_SIZE
            frame_data = video_data[offset:offset + FRAME_SIZE]
            frame_index += 1
        else:
            frame_data = video_data[:FRAME_SIZE]
        try:
            with open(FB_DEVICE, 'wb') as fb:
                fb.write(frame_data)
        except Exception as e:
            print(f"⚠️ 写入帧缓冲失败: {e}")
            break
        time.sleep(frame_duration)
    clear_screen()
    set_backlight(False)
    print("⏹️  视频已停止")
    video_playing = False

def start_playback():
    global video_playing, playback_thread
    if video_playing:
        print("⚠️ 视频已在播放中")
        return
    video_stop_event.clear()
    playback_thread = threading.Thread(target=play_video_loop, daemon=True)
    video_playing = True
    playback_thread.start()

def stop_playback():
    global video_playing, playback_thread
    if not video_playing:
        return
    video_stop_event.set()
    if playback_thread and playback_thread.is_alive():
        playback_thread.join(timeout=2.0)
    video_playing = False

# ========== 指令匹配辅助 ==========
def normalize(text):
    return re.sub(r'\s+', '', text).lower()

def contains_play(text):
    n = normalize(text)
    return 'star' in n and 'tv' in n

def contains_stop(text):
    n = normalize(text)
    return 'stop' in n and 'tv' in n

def contains_exit(text):
    n = normalize(text)
    return 'exit' in n

# ========== 实时指令处理器 ==========
class StableCommandMatcher:
    def __init__(self):
        self.buffer = ""
        self.exit_pending_time = None

    def feed(self, raw_bytes):
        global last_play_time, last_stop_time, running
        try:
            self.buffer += raw_bytes.decode('utf-8', errors='ignore')
        except:
            return

        now = time.time()

        # 播放指令
        if contains_play(self.buffer):
            if now - last_play_time > CMD_COOLDOWN:
                print("📨 [播放] 检测到 star+tv 指令")
                start_playback()
                last_play_time = now
                self.buffer = ""

        # 停止指令（使用 elif 避免同时触发）
        elif contains_stop(self.buffer):
            if now - last_stop_time > CMD_COOLDOWN:
                print("📨 [停止] 检测到 stop+tv 指令")
                stop_playback()
                last_stop_time = now
                self.buffer = ""

        # 退出指令（独立处理）
        if contains_exit(self.buffer):
            # 移除已匹配的 exit 部分
            idx = self.buffer.lower().find('exit')
            if idx != -1:
                self.buffer = self.buffer[idx+4:]

            if self.exit_pending_time is None:
                self.exit_pending_time = now
                print("⏳ 检测到退出指令，请在 2 秒内再次发送 EXIT 以确认退出")
            else:
                if now - self.exit_pending_time <= EXIT_CONFIRM_TIMEOUT:
                    print("✅ 退出指令确认，程序即将退出")
                    running = False
                else:
                    print("⏰ 退出确认超时，请重新发送两次 EXIT")
                    self.exit_pending_time = now

        # 定期清理缓冲区
        if len(self.buffer) > 500:
            self.buffer = self.buffer[-200:]

        # 清理超时等待
        if self.exit_pending_time and (now - self.exit_pending_time) > EXIT_CONFIRM_TIMEOUT:
            self.exit_pending_time = None
            print("⏰ 退出指令等待超时，已取消")

# ========== 软串口接收线程（带冲突恢复） ==========
def open_soft_serial_with_retry(pi, rx_pin, baud, bits, max_attempts=5):
    """尝试打开软串口，如果失败则关闭重试"""
    for attempt in range(max_attempts):
        # 先尝试关闭可能残留的软串口
        try:
            pi.bb_serial_read_close(rx_pin)
            time.sleep(0.2)
        except:
            pass
        try:
            pi.bb_serial_read_open(rx_pin, baud, bits)
            print(f"✅ 软串口接收已启动 (GPIO{rx_pin}, {baud}bps)")
            return True
        except Exception as e:
            print(f"⚠️ 打开软串口失败 (尝试 {attempt+1}/{max_attempts}): {e}")
            time.sleep(1)
    return False

def receiver_thread():
    global running
    matcher = StableCommandMatcher()

    # 尝试打开软串口，失败则退出（让 systemd 重启服务）
    if not open_soft_serial_with_retry(pi, RX_PIN, BAUD, BITS):
        print("❌ 无法打开软串口，接收线程退出")
        return

    try:
        while running:
            (count, data) = pi.bb_serial_read(RX_PIN)
            if count > 0:
                matcher.feed(data)
            else:
                time.sleep(0.005)
    except Exception as e:
        print(f"⚠️ 接收线程异常: {e}")
    finally:
        try:
            pi.bb_serial_read_close(RX_PIN)
            print("🔌 软串口已关闭")
        except:
            pass
        print("📡 接收线程已退出")

# ========== 主程序 ==========
def main():
    global running

    pi.set_mode(TX_PIN, pigpio.OUTPUT)
    pi.set_mode(RX_PIN, pigpio.INPUT)
    pi.set_mode(LED_PIN, pigpio.OUTPUT)
    pi.write(TX_PIN, 1)
    set_backlight(False)
    clear_screen()

    recv_thread = threading.Thread(target=receiver_thread, daemon=True)
    recv_thread.start()

    print("=" * 50)
    print("🎬 视频播放控制器 (稳定版 + GPIO冲突修复)")
    print(f"   使用帧缓冲: {FB_DEVICE}")
    print("   播放指令: 任意位置同时出现 'star' 和 'tv' (不区分大小写/空格)")
    print("   停止指令: 任意位置同时出现 'stop' 和 'tv'")
    print("   退出指令: 2秒内连续两次出现 'exit'")
    print("   ★ 自动处理 GPIO 占用，服务重启无忧 ★")
    print("=" * 50)

    try:
        while running:
            time.sleep(0.2)
    except KeyboardInterrupt:
        print("\n⏹️  手动中断")
        running = False
    finally:
        recv_thread.join(timeout=2.0)
        if video_playing:
            stop_playback()
        # 最后再关闭一次软串口，确保释放
        try:
            pi.bb_serial_read_close(RX_PIN)
        except:
            pass
        pi.stop()
        print("👋 程序结束，所有资源已释放")

if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""将 RGB565 视频文件按位取反，生成新文件"""
import sys
import array

def invert_frame(data):
    arr = array.array('H', data)  # 'H' = 无符号 2 字节（小端）
    for i in range(len(arr)):
        arr[i] ^= 0xFFFF          # 按位取反
    return arr.tobytes()

def main():
    if len(sys.argv) != 3:
        print("用法: python3 invert_video.py <输入.rgb> <输出.rgb>")
        sys.exit(1)
    src = sys.argv[1]
    dst = sys.argv[2]
    print(f"正在取反: {src} -> {dst}")
    with open(src, 'rb') as fin, open(dst, 'wb') as fout:
        chunk_size = 153600  # 一帧 (320*240*2)
        total = 0
        while True:
            chunk = fin.read(chunk_size)
            if not chunk:
                break
            fout.write(invert_frame(chunk))
            total += len(chunk)
            print(f"\r已处理 {total} 字节", end='', flush=True)
    print("\n✅ 取反完成！")

if __name__ == "__main__":
    main()

#!/usr/bin/env python3
# -*- coding: utf-8 -*-

import os
import sys
import time
import glob
import numpy as np
from PIL import Image, ImageDraw, ImageFont

# 帧缓冲设备编号 (通常 fbtft 分配为 fb1)
FB_NUM = 1
FB_DEVICE = f"/dev/fb{FB_NUM}"
SYS_FB = f"/sys/class/graphics/fb{FB_NUM}"

# 默认分辨率（若从 sysfs 读取失败则使用）
DEFAULT_WIDTH = 320
DEFAULT_HEIGHT = 240

# 背光控制相关
BACKLIGHT_PATH = None

def find_backlight_device():
    """查找 fbtft 创建的背光设备路径"""
    candidates = [
        "/sys/class/backlight/st7789v",
        "/sys/class/backlight/fb_st7789v",
        "/sys/class/backlight/spi0.0",
    ]
    for pattern in candidates:
        matches = glob.glob(pattern)
        if matches:
            return matches[0]
    all_devices = glob.glob("/sys/class/backlight/*")
    if all_devices:
        print(f"找到背光设备: {all_devices[0]}")
        return all_devices[0]
    return None

def init_backlight():
    """初始化背光控制，如果可能则设置为最大亮度"""
    global BACKLIGHT_PATH
    BACKLIGHT_PATH = find_backlight_device()
    if not BACKLIGHT_PATH:
        print("警告: 未找到背光设备，背光可能无法调节。请检查 BL 引脚连接。")
        return False

    try:
        with open(os.path.join(BACKLIGHT_PATH, "max_brightness"), "r") as f:
            max_bright = int(f.read().strip())
        if max_bright == 0:
            print("警告: 背光最大亮度为 0，无法调节亮度。背光可能未正确启用。")
            return False
        # 设置为最大亮度
        with open(os.path.join(BACKLIGHT_PATH, "brightness"), "w") as f:
            f.write(str(max_bright))
        print(f"背光设备: {BACKLIGHT_PATH}, 最大亮度: {max_bright}, 已设置为最亮")
        return True
    except Exception as e:
        print(f"背光初始化失败: {e}")
        return False

def set_backlight(brightness_percent):
    """设置背光亮度百分比 (0~100)"""
    if not BACKLIGHT_PATH:
        return
    try:
        with open(os.path.join(BACKLIGHT_PATH, "max_brightness"), "r") as f:
            max_bright = int(f.read().strip())
        if max_bright == 0:
            return
        target = int(max_bright * brightness_percent / 100)
        target = max(0, min(target, max_bright))
        with open(os.path.join(BACKLIGHT_PATH, "brightness"), "w") as f:
            f.write(str(target))
        print(f"背光亮度: {brightness_percent}%")
    except Exception:
        pass  # 静默失败

def get_fb_info():
    """从 sysfs 读取帧缓冲信息: 分辨率, 位深"""
    width = DEFAULT_WIDTH
    height = DEFAULT_HEIGHT
    bpp = 16  # 默认猜测为 16bpp

    # 尝试读取虚拟分辨率
    try:
        with open(os.path.join(SYS_FB, "virtual_size"), "r") as f:
            size = f.read().strip()
            # 格式如 "320,240"
            parts = size.split(',')
            if len(parts) == 2:
                width = int(parts[0])
                height = int(parts[1])
    except:
        pass

    # 尝试读取位深
    try:
        with open(os.path.join(SYS_FB, "bits_per_pixel"), "r") as f:
            bpp = int(f.read().strip())
    except:
        pass

    # 如果 virtual_size 不存在，尝试从 /sys/class/graphics/fb1/modes 读取
    if width == DEFAULT_WIDTH and height == DEFAULT_HEIGHT:
        try:
            with open(os.path.join(SYS_FB, "modes"), "r") as f:
                mode = f.read().strip()
                # 格式如 "U:320x240p-0"
                import re
                match = re.search(r'(\d+)x(\d+)', mode)
                if match:
                    width = int(match.group(1))
                    height = int(match.group(2))
        except:
            pass

    print(f"帧缓冲信息: {width}x{height}, {bpp} bpp")
    return width, height, bpp

def convert_image_to_fb_format(im, width, height, bpp):
    """
    将 PIL Image 对象转换为帧缓冲所需的原始像素数据
    支持 16bpp RGB565 和 32bpp BGRA
    """
    if im.size != (width, height):
        im = im.resize((width, height), Image.Resampling.LANCZOS)

    im = im.convert('RGBA')
    pixels = np.array(im)  # (H, W, 4)

    if bpp == 16:
        # RGB565
        r = (pixels[:, :, 0] >> 3).astype(np.uint16) << 11
        g = (pixels[:, :, 1] >> 2).astype(np.uint16) << 5
        b = (pixels[:, :, 2] >> 3).astype(np.uint16)
        rgb565 = r | g | b
        return rgb565.astype('<u2').tobytes()
    elif bpp == 32:
        # 通常帧缓冲为 BGRA 顺序
        bgra = np.dstack((pixels[:, :, 2], pixels[:, :, 1], pixels[:, :, 0], pixels[:, :, 3]))
        return bgra.tobytes()
    elif bpp == 24:
        return pixels[:, :, :3].tobytes()
    else:
        # 降级到 16bpp
        print(f"不支持的位深 {bpp}，将尝试使用 RGB565 转换")
        r = (pixels[:, :, 0] >> 3).astype(np.uint16) << 11
        g = (pixels[:, :, 1] >> 2).astype(np.uint16) << 5
        b = (pixels[:, :, 2] >> 3).astype(np.uint16)
        rgb565 = r | g | b
        return rgb565.astype('<u2').tobytes()

def draw_test_pattern(width, height):
    """生成测试图像：渐变背景 + 文字 + 彩色条纹"""
    im = Image.new('RGB', (width, height))
    draw = ImageDraw.Draw(im)

    # 横向渐变背景
    for x in range(width):
        r = int(255 * x / width)
        g = int(255 * (width - x) / width)
        b = int(128 * (1 - abs(0.5 - x/width) * 2))
        draw.line([(x, 0), (x, height)], fill=(r, g, b))

    # 彩色矩形条
    bar_height = 40
    colors = [(255,0,0), (0,255,0), (0,0,255), (255,255,0), (255,0,255), (0,255,255)]
    for i, col in enumerate(colors):
        y0 = i * bar_height
        y1 = y0 + bar_height
        draw.rectangle([(0, y0), (width, y1)], fill=col)

    # 文字
    try:
        font = ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", 20)
    except:
        font = ImageFont.load_default()
    draw.text((10, 10), "Raspberry Pi 1A", fill=(255,255,255), font=font)
    draw.text((10, 35), "ST7789 320x240", fill=(255,255,255), font=font)
    draw.text((10, 60), "Framebuffer Test", fill=(255,255,255), font=font)
    draw.text((10, 85), time.strftime("%Y-%m-%d %H:%M:%S"), fill=(255,255,255), font=font)

    return im

def main():
    print("开始 ST7789 屏幕测试...")
    backlight_ok = init_backlight()

    # 检查帧缓冲设备是否存在
    if not os.path.exists(FB_DEVICE):
        print(f"错误: 帧缓冲设备 {FB_DEVICE} 不存在！")
        print("请确认驱动已正确加载，并且 config.txt 中的 dtoverlay 已启用。")
        sys.exit(1)

    # 获取屏幕参数
    width, height, bpp = get_fb_info()
    print(f"使用分辨率: {width}x{height}, 位深: {bpp}")

    try:
        # 打开帧缓冲设备
        fb_fd = os.open(FB_DEVICE, os.O_RDWR)
        # 获取屏幕内存大小 (line_length * virtual_yres)
        # 简单方式：通过 seek 到末尾获取大小
        fb_size = os.lseek(fb_fd, 0, os.SEEK_END)
        os.lseek(fb_fd, 0, os.SEEK_SET)
        print(f"帧缓冲大小: {fb_size} 字节")

        # 生成测试图像并写入
        test_img = draw_test_pattern(width, height)
        fb_data = convert_image_to_fb_format(test_img, width, height, bpp)
        os.write(fb_fd, fb_data)
        print("图像已写入帧缓冲")

        # 如果背光可用，执行呼吸效果；否则直接等待
        if backlight_ok:
            print("开始呼吸灯效果 (亮度循环)，10 秒后继续...")
            start_time = time.time()
            while time.time() - start_time < 10:
                for p in range(0, 101, 5):
                    set_backlight(p)
                    time.sleep(0.03)
                for p in range(100, -1, -5):
                    set_backlight(p)
                    time.sleep(0.03)
            set_backlight(100)
        else:
            print("背光不可调节，跳过呼吸效果。屏幕将保持当前亮度。")
            time.sleep(5)

        # 动态更新时间显示
        print("更新时间显示 5 次...")
        for i in range(5):
            im = draw_test_pattern(width, height)
            draw = ImageDraw.Draw(im)
            draw.text((10, 85), time.strftime("%Y-%m-%d %H:%M:%S"), fill=(255,255,255), font=ImageFont.load_default())
            fb_data = convert_image_to_fb_format(im, width, height, bpp)
            os.lseek(fb_fd, 0, os.SEEK_SET)  # 定位到开头
            os.write(fb_fd, fb_data)
            time.sleep(1)

        print("测试完成，背光保持常亮。")

    except Exception as e:
        print(f"发生错误: {e}")
        import traceback
        traceback.print_exc()
    finally:
        if 'fb_fd' in locals():
            os.close(fb_fd)

if __name__ == "__main__":
    main()

"""
ST7789 显示屏独立测试工具
用法：直接运行，按 Ctrl+C 退出

import pigpio
import time
import sys
import glob
import os

# ========== 配置（与原脚本一致） ==========
def get_last_framebuffer():
    fb_devices = sorted(glob.glob('/dev/fb[0-9]*'))
    return fb_devices[-1] if fb_devices else None

FB_DEVICE = get_last_framebuffer()
if FB_DEVICE is None:
    print("❌ 未找到任何帧缓冲设备 (/dev/fb*)")
    sys.exit(1)
print(f"✅ 使用帧缓冲设备: {FB_DEVICE}")

SCREEN_WIDTH = 320
SCREEN_HEIGHT = 240
BYTES_PER_PIXEL = 2          # RGB565
FRAME_SIZE = SCREEN_WIDTH * SCREEN_HEIGHT * BYTES_PER_PIXEL

# 背光 GPIO（与原脚本一致）
LED_PIN = 18
LED_ON = 1                   # 高电平点亮

# 视频文件路径（可选，若不存在则使用彩条图案）
VIDEO_FILE = "/home/dietpi/test/test_320x240_inv.rgb"

# ========== 连接 pigpiod ==========
def connect_pigpio(host='127.0.0.1', port=8888, retries=10, delay=0.5):
    for i in range(retries):
        pi = pigpio.pi(host, port)
        if pi.connected:
            print(f"✅ 已连接到 pigpiod ({host}:{port})")
            return pi
        pi.stop()
        print(f"⏳ 等待 pigpiod 就绪... ({i+1}/{retries})")
        time.sleep(delay)
    print("❌ 无法连接 pigpiod，请先运行: sudo systemctl start pigpiod")
    sys.exit(1)

pi = connect_pigpio()

# 设置背光引脚为输出
pi.set_mode(LED_PIN, pigpio.OUTPUT)

# ========== 辅助函数 ==========
def set_backlight(on: bool):
    """控制背光"""
    pi.write(LED_PIN, LED_ON if on else (1 - LED_ON))

def clear_screen():
    """清屏为黑色"""
    try:
        with open(FB_DEVICE, 'wb') as fb:
            fb.write(b'\x00' * FRAME_SIZE)
    except Exception as e:
        print(f"⚠️ 清屏失败: {e}")

def write_frame(data):
    """写入一帧数据到帧缓冲"""
    try:
        with open(FB_DEVICE, 'wb') as fb:
            fb.write(data)
    except Exception as e:
        print(f"⚠️ 写入帧缓冲失败: {e}")
        return False
    return True

# ========== 生成测试图案（RGB565） ==========
def generate_color_bar():
    """
    生成彩色条纹图案（RGB565 格式）
    条纹顺序：红、绿、蓝、黄、青、品、白、黑
    """
    # 定义 RGB565 颜色值
    colors = [
        0xF800,  # 红
        0x07E0,  # 绿
        0x001F,  # 蓝
        0xFFE0,  # 黄
        0x07FF,  # 青
        0xF81F,  # 紫
        0xFFFF,  # 白
        0x0000,  # 黑
    ]
    bar_width = SCREEN_WIDTH // len(colors)
    frame = bytearray(FRAME_SIZE)
    for y in range(SCREEN_HEIGHT):
        for x in range(SCREEN_WIDTH):
            idx = x // bar_width
            if idx >= len(colors):
                idx = len(colors)-1
            color = colors[idx]
            # RGB565 小端序写入（与帧缓冲通常要求一致）
            frame[(y*SCREEN_WIDTH + x)*2]   = color & 0xFF
            frame[(y*SCREEN_WIDTH + x)*2+1] = (color >> 8) & 0xFF
    return bytes(frame)

def generate_gradient():
    """生成灰度渐变（RGB565）"""
    frame = bytearray(FRAME_SIZE)
    for y in range(SCREEN_HEIGHT):
        gray = int(255 * y / SCREEN_HEIGHT)
        # 将8位灰度转换为 RGB565 的 R=G=B=gray 的近似值
        r5 = (gray >> 3) & 0x1F
        g6 = (gray >> 2) & 0x3F
        b5 = (gray >> 3) & 0x1F
        rgb565 = (r5 << 11) | (g6 << 5) | b5
        for x in range(SCREEN_WIDTH):
            offset = (y*SCREEN_WIDTH + x)*2
            frame[offset]   = rgb565 & 0xFF
            frame[offset+1] = (rgb565 >> 8) & 0xFF
    return bytes(frame)

def load_video_file():

        return None

def run_test():
    print("=" * 50)
    print("🎨 ST7789 显示屏测试工具")
    print("   背光控制: GPIO{}".format(LED_PIN))
    print("   帧缓冲: {}".format(FB_DEVICE))
    print("   按 Ctrl+C 退出测试")

    # 打开背光

    time.sleep(0.5)

    # 准备测试数据
    video_data = load_video_file()
    if video_data is None:
        # 生成彩条图案（静态）或渐变
        test_pattern = generate_color_bar()
        # 若需要动态效果，可以生成多帧动画，这里简单选择静态彩条
        total_frames = 1
        frame_size = FRAME_SIZE
        data_for_loop = test_pattern
        print("▶️  显示静态彩条图案（按 Ctrl+C 退出）")
    else:
        total_frames = len(video_data) // FRAME_SIZE
        data_for_loop = video_data
        print(f"▶️  循环播放视频 ({total_frames} 帧, 30fps)")

    # 播放循环
    frame_duration = 1.0 / 30  # 30fps
    frame_idx = 0
    try:
        while True:
            if total_frames > 1:
                offset = (frame_idx % total_frames) * FRAME_SIZE

        set_backlight(False)
        pi.write(LED_PIN, 0)   # 确保关闭
